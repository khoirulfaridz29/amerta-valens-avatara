    <!-- SCRIPT LOGIKA APLIKASI -->
    <script>
        // === PASTE KONFIGURASI FIREBASE ANDA DI SINI ===
        const firebaseConfig = {
            apiKey: "API_KEY_ANDA",
            authDomain: "PROJECT_ID.firebaseapp.com",
            projectId: "PROJECT_ID",
            storageBucket: "PROJECT_ID.appspot.com",
            messagingSenderId: "SENDER_ID",
            appId: "APP_ID"
        };

        // Inisialisasi Firebase
        firebase.initializeApp(firebaseConfig);
        const auth = firebase.auth();
        const db = firebase.firestore();

        // Tentukan email mana yang bertindak sebagai Admin / Bos
        const EMAIL_ADMIN = "bos@gmail.com"; 

        // Auto Set Tanggal Hari Ini
        document.getElementById('op-tanggal').valueAsDate = new Date();
        document.getElementById('tr-tanggal').valueAsDate = new Date();

        function hitungLiter() {
            const jerigen = document.getElementById('op-solar-jerigen').value || 0;
            document.getElementById('op-solar-liter').innerText = jerigen * 35;
        }

        // --- AUTHENTICATION ---
        document.getElementById('login-form').addEventListener('submit', (e) => {
            e.preventDefault();
            const email = document.getElementById('login-email').value;
            const pass = document.getElementById('login-password').value;

            auth.signInWithEmailAndPassword(email, pass)
                .catch(err => alert("Gagal login: " + err.message));
        });

        auth.onAuthStateChanged(user => {
            if (user) {
                document.getElementById('login-section').classList.add('hidden');
                document.getElementById('app-section').classList.remove('hidden');

                if (user.email.toLowerCase() === EMAIL_ADMIN.toLowerCase()) {
                    document.getElementById('user-role-badge').innerText = "Akun: Admin / Bos";
                    document.getElementById('admin-dashboard').classList.remove('hidden');
                    loadAdminData();
                } else {
                    document.getElementById('user-role-badge').innerText = "Akun: Operator / Pekerja";
                    document.getElementById('operator-dashboard').classList.remove('hidden');
                    loadOperatorData();
                }
            } else {
                document.getElementById('login-section').classList.remove('hidden');
                document.getElementById('app-section').classList.add('hidden');
                document.getElementById('admin-dashboard').classList.add('hidden');
                document.getElementById('operator-dashboard').classList.add('hidden');
            }
        });

        function logout() {
            auth.signOut();
        }

        // --- OPERATOR LOGIC ---
        document.getElementById('operator-form').addEventListener('submit', (e) => {
            e.preventDefault();
            const jerigen = parseFloat(document.getElementById('op-solar-jerigen').value);
            
            db.collection('laporan_harian').add({
                tanggal: document.getElementById('op-tanggal').value,
                lokasi: document.getElementById('op-lokasi').value,
                hm: parseFloat(document.getElementById('op-hm').value),
                solarJerigen: jerigen,
                solarLiter: jerigen * 35,
                keterangan: document.getElementById('op-keterangan').value,
                operatorEmail: auth.currentUser.email,
                createdAt: firebase.firestore.FieldValue.serverTimestamp()
            }).then(() => {
                alert("Laporan harian berhasil dikirim!");
                document.getElementById('operator-form').reset();
                document.getElementById('op-tanggal').valueAsDate = new Date();
                hitungLiter();
            });
        });

        function loadOperatorData() {
            db.collection('laporan_harian')
                .where('operatorEmail', '==', auth.currentUser.email)
                .orderBy('tanggal', 'desc')
                .onSnapshot(snapshot => {
                    const tbody = document.getElementById('operator-history-table');
                    tbody.innerHTML = '';
                    snapshot.forEach(doc => {
                        const d = doc.data();
                        tbody.innerHTML += `
                            <tr>
                                <td class="p-3">${d.tanggal}</td>
                                <td class="p-3 font-semibold">${d.lokasi}</td>
                                <td class="p-3">${d.hm} HM</td>
                                <td class="p-3">${d.solarJerigen} jrg (${d.solarLiter} L)</td>
                                <td class="p-3 text-slate-500">${d.keterangan || '-'}</td>
                            </tr>
                        `;
                    });
                });
        }

        // --- ADMIN LOGIC ---
        document.getElementById('transaksi-form').addEventListener('submit', (e) => {
            e.preventDefault();
            db.collection('transaksi_kas').add({
                tanggal: document.getElementById('tr-tanggal').value,
                jenis: document.getElementById('tr-jenis').value,
                nominal: parseFloat(document.getElementById('tr-nominal').value),
                keterangan: document.getElementById('tr-keterangan').value,
                createdAt: firebase.firestore.FieldValue.serverTimestamp()
            }).then(() => {
                alert("Transaksi kas disimpan!");
                document.getElementById('transaksi-form').reset();
                document.getElementById('tr-tanggal').valueAsDate = new Date();
            });
        });

        function loadAdminData() {
            // Read Laporan Operator
            db.collection('laporan_harian').orderBy('tanggal', 'desc').onSnapshot(snapshot => {
                const tbody = document.getElementById('admin-operator-table');
                tbody.innerHTML = '';
                snapshot.forEach(doc => {
                    const d = doc.data();
                    tbody.innerHTML += `
                        <tr>
                            <td class="p-3">${d.tanggal}</td>
                            <td class="p-3 font-semibold">${d.lokasi}</td>
                            <td class="p-3">${d.hm} HM</td>
                            <td class="p-3">${d.solarJerigen} Jerigen</td>
                            <td class="p-3 text-amber-600 font-bold">${d.solarLiter} Liter</td>
                            <td class="p-3 text-slate-500">${d.keterangan || '-'}</td>
                        </tr>
                    `;
                });
            });

            // Read Transaksi & Hitung Laba Rugi
            db.collection('transaksi_kas').orderBy('tanggal', 'desc').onSnapshot(snapshot => {
                const tbody = document.getElementById('admin-kas-table');
                tbody.innerHTML = '';
                let totalDebit = 0;
                let totalKredit = 0;

                snapshot.forEach(doc => {
                    const d = doc.data();
                    if (d.jenis === 'debit') totalDebit += d.nominal;
                    if (d.jenis === 'kredit') totalKredit += d.nominal;

                    const isDebit = d.jenis === 'debit';
                    tbody.innerHTML += `
                        <tr>
                            <td class="p-3">${d.tanggal}</td>
                            <td class="p-3">
                                <span class="px-2 py-1 rounded text-xs font-bold ${isDebit ? 'bg-emerald-100 text-emerald-700' : 'bg-rose-100 text-rose-700'}">
                                    ${isDebit ? 'DEBIT (MASUK)' : 'KREDIT (KELUAR)'}
                                </span>
                            </td>
                            <td class="p-3">${d.keterangan}</td>
                            <td class="p-3 font-bold ${isDebit ? 'text-emerald-600' : 'text-rose-600'}">
                                Rp ${d.nominal.toLocaleString('id-ID')}
                            </td>
                        </tr>
                    `;
                });

                // Update Ringkasan Laba Rugi
                document.getElementById('stat-debit').innerText = "Rp " + totalDebit.toLocaleString('id-ID');
                document.getElementById('stat-kredit').innerText = "Rp " + totalKredit.toLocaleString('id-ID');
                
                const laba = totalDebit - totalKredit;
                const labaEl = document.getElementById('stat-laba');
                labaEl.innerText = "Rp " + laba.toLocaleString('id-ID');
                if (laba < 0) {
                    labaEl.parentElement.className = "bg-rose-600 text-white p-5 rounded-xl shadow-sm";
                } else {
                    labaEl.parentElement.className = "bg-blue-600 text-white p-5 rounded-xl shadow-sm";
                }
            });
        }
    </script>
